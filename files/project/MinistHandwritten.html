2. 构造全连接层¶

对于两层全连接层构成的 MLP，假设有 $m$ 个样本，对应损失函数为：$L = \frac{1}{2m}\sum_{i=1}^m(\hat{y}^{(i)}-y^{(i)})^2$，以下推导反向传播的梯度计算：

  • 对于输出层 $\hat{y}^{(i)} = h*w_2+b_2$， $$ \begin{aligned} \frac{\partial L}{\partial w_2} &= \frac{L}{\partial \hat{y}^{(i)}} \cdot \frac{\partial \hat{y}^{(i)}}{\partial w_2} \\ &= \frac{1}{m}\sum_{i=1}^m(\hat{y}^{(i)}-y^{(i)})h^{(i)} \end{aligned} $$ $$ \begin{aligned} \frac{\partial L}{\partial b_2} &= \frac{L}{\partial \hat{y}^{(i)}} \cdot \frac{\partial \hat{y}^{(i)}}{\partial b_2} \\ &= \frac{1}{m}\sum_{i=1}^m(\hat{y}^{(i)}-y^{(i)}) \end{aligned} $$
  • 对于隐藏层 $h^{(i)} = x*w_1+b_1$， $$ \begin{aligned} \frac{\partial L}{\partial w_1} &= \frac{L}{\partial h^{(i)}} \cdot \frac{\partial h^{(i)}}{\partial w_1} \\ &= \frac{1}{m}\sum_{i=1}^m(\hat{y}^{(i)}-y^{(i)})w_2 \cdot x \end{aligned} $$ $$ \begin{aligned} \frac{\partial L}{\partial b_1} &= \frac{L}{\partial h^{(i)}} \cdot \frac{\partial h^{(i)}}{\partial b_1} \\ &= \frac{1}{m}\sum_{i=1}^m(\hat{y}^{(i)}-y^{(i)})w_2 \end{aligned} $$

可以分析得到，输出层给隐藏层传送的梯度应当是 $(\hat{y}^{(i)}-y^{(i)})w_2$，也即线性层向前一层传递的梯度为当前层的梯度乘以当前层的权重矩阵，而计算本层权重梯度和偏差以及取均值操作，都可以统一放在当前层完成。

class LinearLayer:
    """
    Fully connected (dense) layer implementation.
    
    Performs the linear transformation: y = xW + b
    where x is input, W is weight matrix, b is bias vector.
    """
    
    def __init__(self, in_features, out_features):
        """
        Initialize the linear layer.
        
        Args:
            in_features: Number of input features (neurons)
            out_features: Number of output features (neurons)
        """
        self.in_features = in_features
        self.out_features = out_features
        
        # Initialize parameters and gradients dictionaries
        self.params, self.grads = {}, {}
        
        # Initialize weights with random values from standard normal distribution
        self.params['W'] = np.random.randn(in_features, out_features)
        # Initialize biases to zeros
        self.params['b'] = np.zeros(out_features)
        
        # Initialize gradient storage with zeros
        self.grads['W'] = np.zeros_like(self.params['W'])
        self.grads['b'] = np.zeros_like(self.params['b'])
        
        # Cache for storing input during forward pass (used in backward pass)
        self.cache = None
    
    def forward(self, x):
        """
        Forward pass: compute y = xW + b
        
        Args:
            x: Input tensor of shape (batch_size, in_features)
            
        Returns:
            Output tensor of shape (batch_size, out_features)
        """
        self.cache = x  # Store input for backward pass
        return np.dot(x, self.params['W']) + self.params['b']
    
    def backward(self, grad_output):
        """
        Backward pass: compute gradients for parameters and input.
        
        Args:
            grad_output: Gradient from next layer, shape (batch_size, out_features)
            
        Returns:
            Gradient to pass to previous layer, shape (batch_size, in_features)
        """
        x = self.cache
        batch_size = x.shape[0]
        
        # Compute weight gradient: dL/dW = x^T * dL/dy (averaged over batch)
        self.grads['W'] = np.dot(x.T, grad_output) / batch_size
        # Compute bias gradient: dL/db = sum(dL/dy) (averaged over batch)
        self.grads['b'] = np.sum(grad_output, axis=0) / batch_size
        
        # Compute gradient to pass to previous layer: dL/dx = dL/dy * W^T
        grad_input = np.dot(grad_output, self.params['W'].T)
        
        return grad_input
    
    def __repr__(self):
        """String representation of the layer."""
        return f"LinearLayer(in_features={self.in_features}, out_features={self.out_features})"
    
    def num_parameters(self):
        """Return the total number of trainable parameters."""
        return self.in_features * self.out_features + self.out_features

3. 构造 ReLU 激活函数¶

ReLU 激活函数的公式定义为： $$ f(x) = \max(0, x) $$

  • 前向传播过程：$\text{forward}(x) = \max(0, x)$
  • 反向传播过程： $$ \frac{\partial L}{\partial x} = \begin{cases} \frac{\partial L}{\partial f(x)}\frac{f(x)}{x}=\frac{\partial L}{\partial f(x)}, & \text{if } x > 0 \\ \frac{\partial L}{\partial f(x)}\frac{f(x)}{x}=0, & \text{if } x \leq 0 \end{cases} $$

class ReLU:
    """
    Rectified Linear Unit (ReLU) activation function.
    
    Applies element-wise: f(x) = max(0, x)
    """
    
    def __init__(self):
        """Initialize ReLU layer."""
        self.cache = None  # Cache for storing input during forward pass
    
    def forward(self, x):
        """
        Forward pass: apply ReLU activation.
        
        Args:
            x: Input tensor of any shape
            
        Returns:
            Output tensor with same shape as input, with negative values set to 0
        """
        self.cache = x  # Store input for backward pass
        return np.maximum(0, x)
    
    def backward(self, grad_output):
        """
        Backward pass: compute gradient through ReLU.
        
        The gradient is passed through where input > 0, otherwise blocked (set to 0).
        
        Args:
            grad_output: Gradient from next layer
            
        Returns:
            Gradient to pass to previous layer
        """
        x = self.cache
        # Gradient flows through only where input was positive
        grad_input = grad_output * (x > 0)
        return grad_input
    
    def __repr__(self):
        """String representation of the layer."""
        return "ReLU()"
    
    def num_parameters(self):
        """ReLU has no trainable parameters."""
        return 0

4. 构造 Dropout 层¶

Dropout 层在训练过程中以一定概率 p 将输入神经元的输出设为零，以防止过拟合。在测试过程中，Dropout 层不进行任何操作。

前向传播：随机生成一个与输入形状相同的掩码（mask），其中每个元素以概率 1-p 为 1，概率 p 为 0。然后将输入与掩码相乘，并除以 1-p 进行缩放（inverted dropout），保持期望值不变。 $$y = \frac{x \odot \text{mask}}{1-p}$$

反向传播：梯度只通过未被丢弃的神经元传播。对于被丢弃的神经元（mask为0的位置），梯度为0。 $$\frac{\partial L}{\partial x} = \frac{\partial L}{\partial y} \odot \frac{\text{mask}}{1-p}$$ 其中 $\odot$ 表示逐元素乘法。

class Dropout:
    """
    Dropout layer for regularization.
    
    During training, randomly sets elements to zero with probability p.
    Uses inverted dropout: scales output by 1/(1-p) during training
    so no scaling is needed during inference.
    """
    
    def __init__(self, p=0.5):
        """
        Initialize Dropout layer.
        
        Args:
            p: Dropout probability (probability of setting element to zero)
        """
        self.p = p              # Dropout probability
        self.mask = None        # Binary mask for dropped neurons
        self.training = True    # Flag to distinguish training/evaluation mode
    
    def forward(self, x):
        """
        Forward pass: apply dropout during training.
        
        Args:
            x: Input tensor of any shape
            
        Returns:
            Output tensor with same shape, with dropout applied if training
        """
        if self.training:
            # Generate binary mask: 1 with probability (1-p), 0 with probability p
            self.mask = (np.random.rand(*x.shape) > self.p).astype(np.float64)
            # Apply mask and scale by 1/(1-p) to maintain expected value
            return x * self.mask / (1 - self.p)
        else:
            # During evaluation, no dropout is applied
            return x
    
    def backward(self, grad_output):
        """
        Backward pass: gradient only flows through non-dropped neurons.
        
        Args:
            grad_output: Gradient from next layer
            
        Returns:
            Gradient to pass to previous layer
        """
        if self.training:
            # Apply same mask and scaling as forward pass
            return grad_output * self.mask / (1 - self.p)
        else:
            return grad_output
    
    def __repr__(self):
        """String representation of the layer."""
        return f"Dropout(p={self.p})"
    
    def num_parameters(self):
        """Dropout has no trainable parameters."""
        return 0

class TwoLayerMLP:
    """
    Two-layer Multi-Layer Perceptron (MLP) neural network.
    
    Architecture:
        Input -> Linear1 -> ReLU -> Dropout -> Linear2 -> Output
    
    This is a simple feedforward network for classification tasks.
    """
    
    def __init__(self, input_size=784, hidden_size=256, output_size=10, dropout_p=0.5):
        """
        Initialize the MLP network.
        
        Args:
            input_size: Number of input features (784 for flattened 28x28 images)
            hidden_size: Number of neurons in hidden layer
            output_size: Number of output classes (10 for digit classification)
            dropout_p: Dropout probability for regularization
        """
        # Store network dimensions
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        
        # Create network layers
        self.net1 = LinearLayer(input_size, hidden_size)   # First fully connected layer
        self.relu = ReLU()                                  # ReLU activation
        self.dropout = Dropout(p=dropout_p)                 # Dropout for regularization
        self.net2 = LinearLayer(hidden_size, output_size)  # Output layer
        
        # Store all layers in a list for easy iteration
        self.layers = [self.net1, self.relu, self.dropout, self.net2]
        self.training = True  # Flag for training/evaluation mode
    
    def forward(self, x):
        """
        Forward pass through the network.
        
        Args:
            x: Input tensor of shape (batch_size, input_size)
            
        Returns:
            Output logits of shape (batch_size, output_size)
        """
        x = self.net1.forward(x)      # Linear transformation
        x = self.relu.forward(x)      # Non-linear activation
        x = self.dropout.forward(x)   # Dropout regularization
        x = self.net2.forward(x)      # Output layer (logits)
        return x
    
    def backward(self, grad):
        """
        Backward pass through the network (reverse order of forward).
        
        Args:
            grad: Gradient from loss function, shape (batch_size, output_size)
            
        Returns:
            Gradient with respect to input
        """
        grad = self.net2.backward(grad)      # Backprop through output layer
        grad = self.dropout.backward(grad)   # Backprop through dropout
        grad = self.relu.backward(grad)      # Backprop through ReLU
        grad = self.net1.backward(grad)      # Backprop through first layer
        return grad
    
    def train(self):
        """Set the network to training mode."""
        self.training = True
        for layer in self.layers:
            if hasattr(layer, 'training'):
                layer.training = True
    
    def eval(self):
        """Set the network to evaluation mode."""
        self.training = False
        for layer in self.layers:
            if hasattr(layer, 'training'):
                layer.training = False
    
    def get_layers(self):
        """Return list of all layers."""
        return self.layers
    
    def __repr__(self):
        """String representation showing network structure."""
        lines = [f"MLP("]
        for i, layer in enumerate(self.layers):
            lines.append(f"  ({i}): {layer}")
        lines.append(")")
        return "\n".join(lines)
    
    def summary(self):
        """Print detailed network summary including parameter counts."""
        print("=" * 60)
        print(f"{'Layer':<30} {'Output Shape':<15} {'Parameters':<10}")
        print("=" * 60)
        
        total_params = 0
        
        # Define layer information for display
        layer_info = [
            (self.net1, f"(batch, {self.hidden_size})"),
            (self.relu, f"(batch, {self.hidden_size})"),
            (self.dropout, f"(batch, {self.hidden_size})"),
            (self.net2, f"(batch, {self.output_size})")
        ]
        
        # Print each layer's information
        for layer, shape in layer_info:
            params = layer.num_parameters()
            total_params += params
            print(f"{str(layer):<30} {shape:<15} {params:<10}")
        
        print("=" * 60)
        print(f"Total trainable parameters: {total_params:,}")
        print("=" * 60)

6. 损失函数设计¶

6.1 softmax 函数¶

SoftMax 函数用于将模型的输出 logits 转换为概率分布，基本的定义如下： $$ \text{softmax}(z_i) = \frac{e^{z_i}}{\sum_{j=1}^K e^{z_j}} $$

直接计算 $e^z_i$ 可能导致数值溢出，通常会在计算时减去 logits 的最大值。 $$ \text{softmax}(z_i) = \frac{e^{z_i - \max(z)}}{\sum_{j=1}^K e^{z_j - \max(z)}} $$

def softmax(logits):
    """
    Compute softmax probability distribution.
    
    Converts raw logits to probability distribution that sums to 1.
    Uses numerical stability trick: subtract max value before exponentiating.
    
    Formula: softmax(z_i) = exp(z_i - max(z)) / sum(exp(z_j - max(z)))

    Args:
        logits: Raw scores of shape (batch_size, num_classes)
        
    Returns:
        Probability distribution of shape (batch_size, num_classes)
    """
    # Subtract max for numerical stability (prevents overflow)
    exp_logits = np.exp(logits - np.max(logits, axis=1, keepdims=True))
    # Normalize to get probabilities
    probs = exp_logits / np.sum(exp_logits, axis=1, keepdims=True)
    return probs

6.2 交叉熵损失函数¶

交叉熵损失函数用于衡量两个数据分布之间的差异。对于多分类问题，交叉熵定义如下，其中 $K$ 是类别数量： $$ H(y_i, p_i) = -\sum_{i=1}^{K} y_i \log(p_i) $$

  • 前向计算交叉熵损失：$L = -\frac{1}{m}\sum_{i=1}^m\sum_{j=1}^{K} y_{i,j} \log(p_{i,j}) = -\frac{1}{m}\sum_{i=1}^m\log(p_c)$ 其中，$c$ 是真实类别

  • 反向传播梯度：对于单个样本 $L=-\sum_j y_j \log(p_j)$，计算 $L$ 对于类别 $i$ 对应的 logit 参数 $z_i$ 的梯度如下： $$ \frac{\partial L}{\partial z_{i}} = -\sum_j\frac{ y_j}{p_j}\cdot \frac{\partial p_j}{\partial z_i} $$

  • 当 $i=j$ 时，$\frac{\partial p_j}{\partial z_i} = p_i(1 - p_i)$

  • 当 $i \ne j$ 时，$\frac{\partial p_j}{\partial z_i} = -p_j p_i$

因此，合并以上计算后得到： $$ \begin{aligned} \frac{\partial L}{\partial z_{i}} &= -\frac{ y_i}{p_i} \cdot p_i(1 - p_i) + \sum_{j \ne i}\frac{ y_j}{p_j} \cdot p_j p_i \\ &= -y_i(1 - p_i) + \sum_{j \ne i} y_jp_i \\ &= -y_i + p_i(y_i + \sum_{j \neq i} y_j) \\ &\qquad\qquad\quad\underbrace{\qquad\qquad}_{=1} \\ &= p_i - y_i \end{aligned} $$

对于真实类别，$y_i=1$，其他类别 $y_j=0$，因此梯度简化为： $$ \frac{\partial L}{\partial z_{i}} = \begin{cases} p_i - 1, & \text{if } i \text{ is the true class} \\ p_i, & \text{otherwise} \end{cases} $$

对于多个样本，梯度计算为： $$ \frac{\partial L}{\partial z_{i}} = \frac{1}{m}\sum_k^m(p_{k,i} - y_{k,i}) $$

class CrossEntropyLoss:
    """
    Cross-entropy loss function for multi-class classification.
    
    Combines softmax activation and negative log-likelihood loss.
    This is the standard loss function for classification problems.
    
    Formula: L = -1/m * sum(log(p_true_class))
    """
    
    def __init__(self):
        """Initialize the loss function."""
        self.cache = None  # Cache for storing values needed in backward pass
    
    def forward(self, logits, targets):
        """
        Compute the cross-entropy loss.
        
        Args:
            logits: Raw model output of shape (batch_size, num_classes)
            targets: True class labels of shape (batch_size,) as integers
            
        Returns:
            Scalar loss value (averaged over batch)
        """
        batch_size = logits.shape[0]
        
        # Apply softmax to get probabilities
        probs = softmax(logits)
        
        # Extract probability of true class for each sample
        # Add small epsilon (1e-10) to prevent log(0)
        log_probs = -np.log(probs[range(batch_size), targets] + 1e-10)
        
        # Compute mean loss over batch
        loss = np.sum(log_probs) / batch_size
        
        # Cache values for backward pass
        self.cache = (probs, targets, batch_size)
        
        return loss
    
    def backward(self):
        """
        Compute gradient of loss with respect to logits.
        
        The gradient has a simple form: grad = (probs - one_hot_targets) / batch_size
        
        Returns:
            Gradient of shape (batch_size, num_classes)
        """
        probs, targets, batch_size = self.cache
        
        # Start with softmax probabilities
        grad = probs.copy()
        # Subtract 1 from probability at true class position
        # This implements: grad_i = p_i - y_i where y is one-hot encoded
        grad[range(batch_size), targets] -= 1
        # Average over batch
        grad /= batch_size
        
        return grad

7. 随机梯度下降优化器¶

随机梯度下降（SGD）优化器通过在每次迭代中使用一个小批量的数据来近似计算梯度，从而更新模型参数。

  • 参数更新： $$\theta_{t+1} = \theta_t - \eta \nabla_\theta L(\theta_t)$$ 其中 $\theta$ 是模型参数，$\eta$ 是学习率，$\nabla_\theta L(\theta_t)$ 是损失函数对参数的梯度。

  • 动量（Momentum）：为了加速收敛并减少震荡，SGD 通常会引入动量项。动量通过累积之前梯度的指数加权移动平均来更新参数： $$v_t = \beta v_{t-1} - \eta \nabla_\theta L(\theta_t)$$ $$\theta_{t+1} = \theta_t + v_t$$ 其中 $v_t$ 是速度（velocity），$\beta$ 是动量系数（通常设为 0.9）。动量项帮助参数更新在相关方向上加速，同时在不相关方向上减速。

  • 梯度清零：在每次参数更新后，需要将梯度清零，以便为下一次迭代做准备。这是因为梯度是累积计算的，如果不清零，会导致梯度累加，影响训练效果。

class SGD:
    """
    Stochastic Gradient Descent optimizer with optional momentum.
    
    Updates parameters using the rule:
        v = momentum * v - learning_rate * gradient
        param = param + v
    
    When momentum=0, this reduces to vanilla SGD:
        param = param - learning_rate * gradient
    """
    
    def __init__(self, learning_rate=0.01, momentum=0.0):
        """
        Initialize the SGD optimizer.
        
        Args:
            learning_rate: Step size for parameter updates
            momentum: Momentum coefficient (0 to 1, typically 0.9)
        """
        self.learning_rate = learning_rate
        self.momentum = momentum
        self.velocity = {}  # Store velocity for each parameter
    
    def step(self, layers):
        """
        Perform one optimization step (update all parameters).
        
        Args:
            layers: List of layers containing parameters to update
        """
        for i, layer in enumerate(layers):
            # Only process layers that have parameters
            if hasattr(layer, 'params') and len(layer.params) > 0:
                for key in layer.params:
                    # Initialize velocity to zeros on first call
                    if (i, key) not in self.velocity:
                        self.velocity[(i, key)] = np.zeros_like(layer.params[key])
                    
                    # Update velocity: v = momentum * v - lr * gradient
                    self.velocity[(i, key)] = (
                        self.momentum * self.velocity[(i, key)] - 
                        self.learning_rate * layer.grads[key]
                    )
                    
                    # Update parameter: param = param + velocity
                    layer.params[key] += self.velocity[(i, key)]

    def zero_grad(self, layers):
        """
        Reset all gradients to zero.
        
        This should be called before each backward pass to prevent
        gradient accumulation from previous iterations.
        
        Args:
            layers: List of layers containing gradients to reset
        """
        for layer in layers:
            if hasattr(layer, 'grads'):
                for key in layer.grads:
                    layer.grads[key] = np.zeros_like(layer.grads[key])

def train_epoch(epoch, n_epochs, model, train_loader, criterion, optimizer):
    """
    Train the model for one complete epoch.
    
    An epoch consists of one pass through the entire training dataset.
    
    Args:
        epoch: Current epoch number (0-indexed)
        n_epochs: Total number of epochs
        model: Neural network model to train
        train_loader: DataLoader for training data
        criterion: Loss function
        optimizer: Optimizer for parameter updates
    
    Returns:
        avg_loss: Average loss over the epoch
        avg_acc: Average accuracy over the epoch
    """
    # Set model to training mode (enables dropout)
    model.train()
    
    # Initialize counters for statistics
    total_loss, total_correct, total_samples = 0.0, 0, 0
    num_batches = len(train_loader)
    
    # Iterate over all batches in the training set
    for batch_idx, (data, target) in enumerate(train_loader):
        # Convert PyTorch tensors to NumPy arrays
        data, target, batch_size = data.numpy(), target.numpy(), data.shape[0]
        # Flatten images from (batch, 1, 28, 28) to (batch, 784)
        input = data.reshape(data.shape[0], -1)
        
        # === Forward Pass ===
        output = model.forward(input)           # Get model predictions
        loss = criterion.forward(output, target)  # Compute loss

        # === Backward Pass ===
        grad = criterion.backward()                    # Get gradient from loss
        last_grad = model.backward(grad).reshape(data.shape)  # Backpropagate
        
        # === Parameter Update ===
        layers = model.get_layers()
        optimizer.step(layers)        # Update parameters using gradients
        optimizer.zero_grad(layers)   # Reset gradients for next iteration
        
        # === Compute Accuracy ===
        correct = np.sum(np.argmax(output, axis=1) == target)
        
        # === Accumulate Statistics ===
        total_loss += loss * batch_size
        total_correct += correct
        total_samples += batch_size
        
        # Print progress every 100 batches
        if (batch_idx + 1) % 100 == 0:
            print(f'[Epoch {epoch + 1}/{n_epochs}] '
                  f'[Batch {batch_idx + 1}/{num_batches}] '
                  f'[Loss {loss:.4f}] '
                  f'[Acc {100.0 * correct / batch_size:.2f}%] '
                  f'[Last Grad {last_grad.mean():.6f}]')
    
    # Compute epoch averages
    avg_loss = total_loss / total_samples
    avg_acc = total_correct / total_samples
    
    return avg_loss, avg_acc


def evaluate(model, test_loader):
    """
    Evaluate the model on a test/validation set.
    
    Args:
        model: Neural network model to evaluate
        test_loader: DataLoader for test data
    
    Returns:
        accuracy: Classification accuracy (0 to 1)
    """
    # Set model to evaluation mode (disables dropout)
    model.eval()
    
    total_correct, total_samples = 0, 0
    
    # Iterate over all test batches
    for data, target in test_loader:
        # Convert to NumPy and flatten
        data, target = data.numpy(), target.numpy()
        input = data.reshape(data.shape[0], -1)
        
        # Forward pass only (no gradient computation needed)
        output = model.forward(input)
        
        # Get predicted classes
        predictions = np.argmax(output, axis=1)
        
        # Count correct predictions
        total_correct += np.sum(predictions == target)
        total_samples += data.shape[0]
    
    return total_correct / total_samples

def plot_training_history(meter, save_path=None):
    """
    Visualize training history with loss and accuracy curves.
    
    Creates a two-panel figure showing:
    - Left: Training loss over epochs
    - Right: Training and test accuracy over epochs
    
    Args:
        meter: Dictionary containing 'train_loss', 'train_acc', 'test_acc'
        save_path: Optional file path to save the figure
    """
    # Create epoch range for x-axis
    epochs = range(1, len(meter['train_loss']) + 1)
    
    # Create figure with two subplots
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    # === Plot 1: Training Loss ===
    axes[0].plot(epochs, meter['train_loss'], 'b-o', 
                 label='Train Loss', linewidth=2, markersize=6)
    axes[0].set_xlabel('Epoch', fontsize=12)
    axes[0].set_ylabel('Loss', fontsize=12)
    axes[0].set_title('Training Loss over Epochs', fontsize=14)
    axes[0].legend(fontsize=10)
    axes[0].grid(True, linestyle='--', alpha=0.7)
    
    # === Plot 2: Accuracy Curves ===
    # Convert accuracy to percentage for display
    axes[1].plot(epochs, [acc * 100 for acc in meter['train_acc']], 'b-o', 
                 label='Train Accuracy', linewidth=2, markersize=6)
    axes[1].plot(epochs, [acc * 100 for acc in meter['test_acc']], 'r-s', 
                 label='Test Accuracy', linewidth=2, markersize=6)
    axes[1].set_xlabel('Epoch', fontsize=12)
    axes[1].set_ylabel('Accuracy (%)', fontsize=12)
    axes[1].set_title('Training and Test Accuracy over Epochs', fontsize=14)
    axes[1].legend(fontsize=10)
    axes[1].grid(True, linestyle='--', alpha=0.7)
    
    # Adjust layout to prevent overlap
    plt.tight_layout()
    
    # Save figure if path provided
    if save_path:
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        print(f"Figure saved to {save_path}")
    
    plt.show()
    
    # === Print Summary Statistics ===
    print(f"\n{'='*50}")
    print(f"Training Summary:")
    print(f"  Final Train Loss: {meter['train_loss'][-1]:.4f}")
    print(f"  Final Train Accuracy: {meter['train_acc'][-1]*100:.2f}%")
    print(f"  Final Test Accuracy: {meter['test_acc'][-1]*100:.2f}%")
    # Find best test accuracy and corresponding epoch
    best_acc = max(meter['test_acc'])
    best_epoch = meter['test_acc'].index(best_acc) + 1
    print(f"  Best Test Accuracy: {best_acc*100:.2f}% (Epoch {best_epoch})")
    print(f"{'='*50}")

def visualize_predictions(model, test_loader, num_samples=16, save_path=None):
    """
    Visualize model predictions on random test samples.
    
    Displays a grid of test images with their true labels, predicted labels,
    and prediction confidence. Correct predictions are shown in green,
    incorrect ones in red.
    
    Args:
        model: Trained neural network model
        test_loader: DataLoader for test data
        num_samples: Number of samples to display
        save_path: Optional file path to save the figure
    """
    # Set model to evaluation mode
    model.eval()
    
    # Get a batch of test data
    data_iter = iter(test_loader)
    images, labels = next(data_iter)
    images, labels = images.numpy(), labels.numpy()
    
    # === Forward Pass ===
    inputs = images.reshape(images.shape[0], -1)  # Flatten images
    outputs = model.forward(inputs)
    probs = softmax(outputs)                       # Convert to probabilities
    predictions = np.argmax(outputs, axis=1)       # Get predicted classes
    confidences = np.max(probs, axis=1)            # Get confidence scores
    
    # === Setup Figure Grid ===
    num_samples = min(num_samples, len(images))
    cols = 4
    rows = (num_samples + cols - 1) // cols  # Ceiling division
    
    fig, axes = plt.subplots(rows, cols, figsize=(12, 3 * rows))
    axes = axes.flatten()
    
    # === Plot Each Sample ===
    for i in range(num_samples):
        ax = axes[i]
        img = images[i].squeeze()  # Remove channel dimension
        
        ax.imshow(img, cmap='gray')
        
        true_label = labels[i]
        pred_label = predictions[i]
        conf = confidences[i] * 100
        
        # Color code based on correctness
        if true_label == pred_label:
            color = 'green'
            title = f'True: {true_label}, Pred: {pred_label}\nConf: {conf:.1f}%'
        else:
            color = 'red'
            title = f'True: {true_label}, Pred: {pred_label} ✗\nConf: {conf:.1f}%'
        
        ax.set_title(title, fontsize=10, color=color)
        ax.axis('off')
    
    # Hide unused subplots
    for i in range(num_samples, len(axes)):
        axes[i].axis('off')
    
    plt.suptitle('Model Predictions on Test Samples', fontsize=14, fontweight='bold')
    plt.tight_layout()
    
    if save_path:
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        print(f"Figure saved to {save_path}")
    
    plt.show()
    
    # Print accuracy on displayed samples
    correct = np.sum(predictions[:num_samples] == labels[:num_samples])
    print(f"\nAccuracy on displayed samples: {correct}/{num_samples} ({100*correct/num_samples:.1f}%)")


def visualize_misclassified(model, test_loader, num_samples=12, save_path=None):
    """
    Visualize samples that the model misclassified.
    
    Helps identify patterns in model errors and difficult examples.
    
    Args:
        model: Trained neural network model
        test_loader: DataLoader for test data
        num_samples: Number of misclassified samples to show
        save_path: Optional file path to save the figure
    """
    # Set model to evaluation mode
    model.eval()
    
    # Storage for misclassified samples
    misclassified_images = []
    misclassified_labels = []
    misclassified_preds = []
    misclassified_confs = []
    
    # === Collect Misclassified Samples ===
    for data, target in test_loader:
        data, target = data.numpy(), target.numpy()
        inputs = data.reshape(data.shape[0], -1)
        
        # Forward pass
        outputs = model.forward(inputs)
        probs = softmax(outputs)
        predictions = np.argmax(outputs, axis=1)
        confidences = np.max(probs, axis=1)
        
        # Find indices of misclassified samples
        wrong_mask = predictions != target
        wrong_indices = np.where(wrong_mask)[0]
        
        # Store misclassified samples
        for idx in wrong_indices:
            misclassified_images.append(data[idx])
            misclassified_labels.append(target[idx])
            misclassified_preds.append(predictions[idx])
            misclassified_confs.append(confidences[idx])
            
            # Stop once we have enough samples
            if len(misclassified_images) >= num_samples:
                break
        
        if len(misclassified_images) >= num_samples:
            break
    
    # === Visualize Misclassified Samples ===
    num_samples = min(num_samples, len(misclassified_images))
    if num_samples == 0:
        print("No misclassified samples found!")
        return
    
    # Setup figure grid
    cols = 4
    rows = (num_samples + cols - 1) // cols
    
    fig, axes = plt.subplots(rows, cols, figsize=(12, 3 * rows))
    axes = axes.flatten() if num_samples > 1 else [axes]
    
    # Plot each misclassified sample
    for i in range(num_samples):
        ax = axes[i]
        img = misclassified_images[i].squeeze()
        
        ax.imshow(img, cmap='gray')
        
        title = (f'True: {misclassified_labels[i]}, Pred: {misclassified_preds[i]}\n'
                f'Conf: {misclassified_confs[i]*100:.1f}%')
        ax.set_title(title, fontsize=10, color='red')
        ax.axis('off')
    
    # Hide unused subplots
    for i in range(num_samples, len(axes)):
        axes[i].axis('off')
    
    plt.suptitle('Misclassified Samples', fontsize=14, fontweight='bold', color='red')
    plt.tight_layout()
    
    if save_path:
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        print(f"Figure saved to {save_path}")
    
    plt.show()
